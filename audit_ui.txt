# Sidebar
with st.sidebar:
    st.markdown('<p class="sidebar-title">🔐 Access Control</p>', unsafe_allow_html=True)
    user_id = st.selectbox("User", [
        "admin_user", "hr_user", "finance_user", "legal_user", "test_user"
    ], label_visibility="collapsed")

    department = st.selectbox("Department", [
        "general", "hr", "finance", "legal"
    ], label_visibility="collapsed")

    top_k = st.slider("Chunks", 1, 10, 5)

    st.markdown("---")
    st.markdown('<p class="sidebar-title">📋 Role Permissions</p>', unsafe_allow_html=True)

    role_data = {
        "admin_user": ("admin", "All departments", "#4ade80"),
        "hr_user": ("hr_staff", "HR · General", "#93c5fd"),
        "finance_user": ("finance_staff", "Finance · General", "#93c5fd"),
        "legal_user": ("legal_staff", "Legal · General", "#93c5fd"),
        "test_user": ("employee", "General only", "#f87171"),
    }
    role, access, color = role_data.get(user_id, ("unknown", "unknown", "#94a3b8"))
    st.markdown(f"""
<div style="background:#0a1020;border:1px solid #1e2a4a;border-radius:8px;padding:0.8rem;margin-top:0.5rem;">
    <div style="font-size:0.75rem;color:#64748b;">Current user</div>
    <div style="font-size:1rem;color:#f1f5f9;font-weight:600;margin:0.2rem 0;">{user_id}</div>
    <div style="font-size:0.75rem;color:#64748b;">Role: <span style="color:{color};font-weight:600;">{role}</span></div>
    <div style="font-size:0.75rem;color:#64748b;margin-top:0.2rem;">Access: <span style="color:#94a3b8;">{access}</span></div>
</div>
""", unsafe_allow_html=True)

    st.markdown("---")
    st.markdown('<p class="sidebar-title">📤 Ingest Document</p>', unsafe_allow_html=True)
    uploaded_file = st.file_uploader("Upload", type=["pdf", "docx", "xlsx", "txt", "csv"], label_visibility="collapsed")
    ingest_dept = st.selectbox("To department", ["general", "hr", "finance", "legal"])
    if st.button("⬆ Ingest Document", use_container_width=True) and uploaded_file:
        with st.spinner("Ingesting..."):
            response = requests.post(
                f"{API_URL}/ingest",
                files={"file": (uploaded_file.name, uploaded_file, "application/octet-stream")},
                data={"department": ingest_dept}
            )
            if response.status_code == 200:
                result = response.json()
                st.success(f"✅ {result['chunks_created']} chunks ingested from {uploaded_file.name}")
            else:
                st.error(f"❌ {response.text}")

# Chat history
if "messages" not in st.session_state:
    st.session_state.messages = []

for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"], unsafe_allow_html=True)

# Query
if prompt := st.chat_input("Ask anything about your documents..."):
    st.session_state.messages.append({"role": "user", "content": prompt})
    with st.chat_message("user"):
        st.markdown(prompt)

    with st.chat_message("assistant"):
        with st.spinner("Searching documents and generating answer..."):
            try:
                response = requests.post(
                    f"{API_URL}/query",
                    json={
                        "query": prompt,
                        "department": department,
                        "user_id": user_id,
                        "top_k": top_k
                    }
                )
                data = response.json()
                answer = data.get("answer", "No answer returned.")
                sources = data.get("sources", [])
                cost = data.get("cost", {})
                model = data.get("model", "unknown")
                chunks_used = data.get("chunks_used", 0)
                total_cost = cost.get("total_cost_usd", 0)
                total_tokens = cost.get("total_tokens", 0)

                if "Access denied" in answer:
                    html = f"""
<div class="denied-container">
    <div class="denied-label">🚫 Access Denied</div>
    <div class="denied-text">{answer}</div>
</div>"""
                    st.markdown(html, unsafe_allow_html=True)
                    full_response = html

                elif "do not meet the relevance threshold" in answer or "not enough information" in answer.lower():
                    html = f"""
<div class="warning-container">
    <div class="warning-label">⚠ No Relevant Documents Found</div>
    <div class="warning-text">{answer}</div>
</div>"""
                    st.markdown(html, unsafe_allow_html=True)
                    full_response = html

                else:
                    sources_html = ""
                    if sources:
                        sources_html = '<div class="sources-label">📚 Sources</div>'
                        for s in sources:
                            score = s.get("score", 0)
                            filename = s.get("filename", "unknown")
                            preview = s.get("preview", "")
                            score_pct = int(score * 100)

                            if score >= 0.8:
                                pill_class = "score-high"
                                bar_color = "#4ade80"
                            elif score >= 0.5:
                                pill_class = "score-mid"
                                bar_color = "#fbbf24"
                            else:
                                pill_class = "score-low"
                                bar_color = "#f87171"

                            sources_html += f"""
<div class="source-card">
    <div class="source-header">
        <span class="source-filename">📄 {filename}</span>
        <span class="score-pill {pill_class}">{score_pct}% confidence</span>
    </div>
    <div class="score-bar-bg">
        <div style="background:{bar_color};height:4px;border-radius:4px;width:{score_pct}%;"></div>
    </div>
    <div class="source-preview">{preview}</div>
</div>"""

                    cost_html = f"""
<div class="cost-bar">
    <div class="cost-item">Model <span>{model}</span></div>
    <div class="cost-item">Chunks <span>{chunks_used}</span></div>
    <div class="cost-item">Tokens <span>{total_tokens:,}</span></div>
    <div class="cost-item">Cost <span>${total_cost:.6f}</span></div>
</div>"""

                    html = f"""
<div class="answer-container">
    <div class="answer-label">Answer</div>
    <div class="answer-text">{answer}</div>
</div>
{sources_html}
{cost_html}"""
                    st.markdown(html, unsafe_allow_html=True)
                    full_response = html

            except Exception as e:
                st.error(f"Connection error: {e}")
                full_response = f"Error: {e}"

    st.session_state.messages.append({"role": "assistant", "content": full_response})
